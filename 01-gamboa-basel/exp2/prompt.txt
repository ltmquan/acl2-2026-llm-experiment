(in-package "ACL2")

(include-book "nonstd/nsa/trig" :dir :system)

(defun basel-series (i n)
  (declare (xargs :measure (nfix (- n i))))
  (if (zp (- n i))
      0
      (+ (expt i -2)
         (basel-series (1+ i) n))))
